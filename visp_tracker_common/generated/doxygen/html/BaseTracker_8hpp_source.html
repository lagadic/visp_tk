145 template <typename Type>
146 void display_overlay_features(const VISP_NAMESPACE_ADDRESSING vpImage<Type> &I)
147 {
148 const VISP_NAMESPACE_ADDRESSING vpColor colors[] = { VISP_NAMESPACE_ADDRESSING vpColor::green, VISP_NAMESPACE_ADDRESSING vpColor::orange,
149 VISP_NAMESPACE_ADDRESSING vpColor::cyan, VISP_NAMESPACE_ADDRESSING vpColor::purple,
150 VISP_NAMESPACE_ADDRESSING vpColor::yellow };
151 const unsigned int nb_colors = sizeof(colors) / sizeof(colors[0]);
152 const unsigned int v_offset = 20;
153 const rclcpp::Time now = this->now();
154 std::scoped_lock lock(m_mutex_overlay_features);
155 unsigned int idx = 0;
156 for (const auto &overlay : m_overlay_features) {
157 // Features not received recently are not displayed anymore
158 if ((!overlay) || ((now - overlay->first).seconds() > s_overlay_features_max_age)) {
159 continue;
160 }
161 for (const auto &feature : overlay->second.features) {
162 const VISP_NAMESPACE_ADDRESSING vpColor &color = colors[idx % nb_colors];
163 display::displayNamedFeature(I, feature, color, 2, display::PointsType::CROSS);
164 if (!feature.name.empty()) {
165 VISP_NAMESPACE_ADDRESSING vpDisplay::displayText(I, static_cast<int>(I.getHeight()) - static_cast<int>(v_offset * (idx + 1)), 10,
166 feature.name, color);
167 }
168 ++idx;
169 }
170 }
171 }
